# Админка заявок Velaro — план реализации

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** закрытая админка `/admin` для владельца и менеджеров: вход по логину и паролю, список заявок, карточка с назначением, статусами, заметками и историей, управление менеджерами, ссылка на карточку в Telegram.

**Architecture:** Next.js App Router (уже есть в проекте). Вся логика — в `lib/auth/*` и `lib/admin/*`, функции принимают `db: AppDb` параметром и тестируются на PGlite. Страницы `/admin/*` — серверные компоненты; изменения — серверные функции (`app/admin/actions.ts`), каждая сама проверяет сессию и роль. Сессия — случайный токен в httpOnly-cookie, в базе хранится только его SHA-256.

**Tech Stack:** Next.js 15.5, React 19 (`useActionState`), TypeScript, Drizzle ORM 0.45 + pg, Vitest 3 + PGlite, `node:crypto` (scrypt, sha256).

**Spec:** `docs/superpowers/specs/2026-10-16-admin-design.md`

## Global Constraints

- Никаких новых npm-зависимостей: пароли — `scrypt` из `node:crypto`.
- Только стандартный Postgres (база переедет в РФ): без Neon-специфики.
- Статусы: `new`, `in_progress`, `confirmed`, `completed`, `cancelled`; подписи: Новая, В работе, Подтверждена, Завершена, Отменена.
- Роли: `owner`, `manager`. Владельца нельзя отключить, понизить, удалить; удаления аккаунтов нет вообще.
- Ошибка входа всегда «Неверный логин или пароль.»; блокировка — 5 неудачных попыток за 15 минут на пару логин+IP: «Слишком много попыток. Попробуйте через 15 минут.»
- Сессия 30 дней, продлевается, если осталось меньше 15 дней. Cookie `velaro_session`: HttpOnly, Secure в продакшене, SameSite=Lax, Path=/admin.
- Логин: `^[a-z0-9._-]{3,32}$` после trim+lowercase. Пароль: 8–200 символов. Имя: 1–60 символов. Заметка: 1–2000 символов.
- Права проверяются на сервере в каждой серверной функции, не только скрытием кнопок.
- Дизайн как у лендинга: Archivo (крупные заголовки), Inter (текст), JetBrains Mono (только шапка), белый фон, `#141414`, скругления 16/24 px, пилюли.
- Все тексты интерфейса — по-русски, без точки в конце заголовков.
- Коммиты: автор `gojoworkk <194110524+gojoworkk@users.noreply.github.com>` (уже настроен в репозитории), в конце сообщения строка `Co-Authored-By: Claude Opus 5.5 <noreply@anthropic.com>`.
- Ветка: `feature/admin`.

import { randomBytes, scrypt as scryptCb, timingSafeEqual } from 'node:crypto';

const N = 16384;
const R = 8;
const P = 1;
const KEYLEN = 64;
const MAXMEM = 64 * 1024 * 1024;

function scrypt(password: string, salt: Buffer, keylen: number, n: number, r: number, p: number): Promise<Buffer> {
  return new Promise((resolve, reject) =>
    scryptCb(password.normalize('NFKC'), salt, keylen, { N: n, r, p, maxmem: MAXMEM }, (err, key) =>
      err ? reject(err) : resolve(key),
    ),
  );
}

/** Формат: scrypt$N$r$p$соль$хэш (base64) — параметры хранятся рядом, их можно будет усилить */
export async function hashPassword(password: string): Promise<string> {
  const salt = randomBytes(16);
  const hash = await scrypt(password, salt, KEYLEN, N, R, P);
  return ['scrypt', N, R, P, salt.toString('base64'), hash.toString('base64')].join('$');
}

export async function verifyPassword(password: string, stored: string): Promise<boolean> {
  const parts = stored.split('$');
  if (parts.length !== 6 || parts[0] !== 'scrypt') return false;
  const [, n, r, p, saltB64, hashB64] = parts;
  const expected = Buffer.from(hashB64, 'base64');
  if (expected.length === 0) return false;
  try {
    const actual = await scrypt(password, Buffer.from(saltB64, 'base64'), expected.length, Number(n), Number(r), Number(p));
    return timingSafeEqual(actual, expected);
  } catch {
    return false;
  }
}

import { users, type Role } from '@/db/schema';
import type { AppDb } from '@/db/types';
import { hashPassword } from '@/lib/auth/password';
import type { SessionUser } from '@/lib/auth/session';

let cachedHash: Promise<string> | null = null;

/** Пользователь в тестовой базе; пароль по умолчанию «password123» */
export async function makeUser(
  db: AppDb,
  over: { login?: string; name?: string; role?: Role; active?: boolean; mustChangePassword?: boolean; password?: string } = {},
): Promise<SessionUser> {
  cachedHash ??= hashPassword('password123');
  const passwordHash = over.password ? await hashPassword(over.password) : await cachedHash;
  const [u] = await db
    .insert(users)
    .values({
      login: over.login ?? 'anna',
      name: over.name ?? 'Анна',
      role: over.role ?? 'manager',
      active: over.active ?? true,
      mustChangePassword: over.mustChangePassword ?? false,
      passwordHash,
    })
    .returning();
  return { id: u.id, login: u.login, name: u.name, role: u.role, mustChangePassword: u.mustChangePassword };
}

import { createHash, randomBytes } from 'node:crypto';
import { and, eq, ne } from 'drizzle-orm';
import { sessions, users, type Role } from '@/db/schema';
import type { AppDb } from '@/db/types';

const DAY = 86_400_000;
export const SESSION_TTL_MS = 30 * DAY;
const RENEW_BELOW_MS = 15 * DAY;

export type SessionUser = { id: number; login: string; name: string; role: Role; mustChangePassword: boolean };

export const hashToken = (token: string) => createHash('sha256').update(token).digest('hex');

/** Создаёт сессию и возвращает токен для cookie; в базе остаётся только его хэш */
export async function createSession(db: AppDb, userId: number, now: Date): Promise<string> {
  const token = randomBytes(32).toString('base64url');
  await db.insert(sessions).values({
    id: hashToken(token),
    userId,
    createdAt: now,
    expiresAt: new Date(now.getTime() + SESSION_TTL_MS),
  });
  return token;
}

export async function getSessionUser(db: AppDb, token: string | null | undefined, now: Date): Promise<SessionUser | null> {
  if (!token) return null;
  const id = hashToken(token);
  const [row] = await db
    .select({
      expiresAt: sessions.expiresAt,
      active: users.active,
      id: users.id,
      login: users.login,
      name: users.name,
      role: users.role,
      mustChangePassword: users.mustChangePassword,
    })
    .from(sessions)
    .innerJoin(users, eq(users.id, sessions.userId))
    .where(eq(sessions.id, id));
  if (!row) return null;
  if (!row.active || row.expiresAt.getTime() <= now.getTime()) {
    await db.delete(sessions).where(eq(sessions.id, id));
    return null;
  }
  if (row.expiresAt.getTime() - now.getTime() < RENEW_BELOW_MS) {
    await db.update(sessions).set({ expiresAt: new Date(now.getTime() + SESSION_TTL_MS) }).where(eq(sessions.id, id));
  }
  return { id: row.id, login: row.login, name: row.name, role: row.role, mustChangePassword: row.mustChangePassword };
}

export async function deleteSession(db: AppDb, token: string): Promise<void> {
  await db.delete(sessions).where(eq(sessions.id, hashToken(token)));
}

/** Выкидывает пользователя со всех устройств (кроме текущей сессии, если она передана) */
export async function deleteUserSessions(db: AppDb, userId: number, exceptToken?: string): Promise<void> {
  await db
    .delete(sessions)
    .where(exceptToken ? and(eq(sessions.userId, userId), ne(sessions.id, hashToken(exceptToken))) : eq(sessions.userId, userId));
}

import { asc, eq } from 'drizzle-orm';
import { users, type Role } from '@/db/schema';
import type { AppDb } from '@/db/types';
import { hashPassword } from '@/lib/auth/password';
import { deleteUserSessions, type SessionUser } from '@/lib/auth/session';

export type UserResult = { ok: true } | { ok: false; errors: Record<string, string> };
export type UserRow = { id: number; login: string; name: string; role: Role; active: boolean; lastLoginAt: Date | null };
export type NewUser = { login: string; name: string; password: string };

const LOGIN_RE = /^[a-z0-9._-]{3,32}$/;
const OK: UserResult = { ok: true };
const fail = (errors: Record<string, string>): UserResult => ({ ok: false, errors });
const FORBIDDEN = fail({ form: 'Недостаточно прав.' });

export const normalizeLogin = (s: string) => s.trim().toLowerCase();

function passwordError(p: string): string | null {
  if (p.length < 8) return 'Пароль — минимум 8 символов.';
  if (p.length > 200) return 'Слишком длинный пароль.';
  return null;
}

async function insertUser(db: AppDb, input: NewUser, role: Role, mustChangePassword: boolean): Promise<UserResult> {
  const login = normalizeLogin(input.login);
  const name = input.name.trim();
  const errors: Record<string, string> = {};
  if (!LOGIN_RE.test(login)) errors.login = 'Логин: 3–32 символа — латиница, цифры, точка, дефис, подчёркивание.';
  if (!name || name.length > 60) errors.name = 'Укажите имя (до 60 символов).';
  const pe = passwordError(input.password);
  if (pe) errors.password = pe;
  if (Object.keys(errors).length) return fail(errors);

  const [taken] = await db.select({ id: users.id }).from(users).where(eq(users.login, login));
  if (taken) return fail({ login: 'Такой логин уже занят.' });
  await db.insert(users).values({ login, name, role, mustChangePassword, passwordHash: await hashPassword(input.password) });
  return OK;
}

/** Первый и единственный владелец — создаётся скриптом из терминала */
export async function createOwner(db: AppDb, input: NewUser): Promise<UserResult> {
  const [owner] = await db.select({ id: users.id }).from(users).where(eq(users.role, 'owner'));
  if (owner) return fail({ form: 'Владелец уже создан.' });
  return insertUser(db, input, 'owner', false);
}

export async function createManager(db: AppDb, actor: SessionUser, input: NewUser): Promise<UserResult> {
  if (actor.role !== 'owner') return FORBIDDEN;
  return insertUser(db, input, 'manager', true);
}

export async function listUsers(db: AppDb): Promise<UserRow[]> {
  return db
    .select({ id: users.id, login: users.login, name: users.name, role: users.role, active: users.active, lastLoginAt: users.lastLoginAt })
    .from(users)
    .orderBy(asc(users.id));
}

/** Менеджер, которым может управлять владелец; владельца трогать нельзя */
async function manageable(db: AppDb, actor: SessionUser, userId: number): Promise<UserResult> {
  if (actor.role !== 'owner') return FORBIDDEN;
  const [u] = await db.select({ role: users.role }).from(users).where(eq(users.id, userId));
  if (!u) return fail({ form: 'Сотрудник не найден.' });
  if (u.role === 'owner') return fail({ form: 'Владельца изменить нельзя.' });
  return OK;
}

export async function resetPassword(db: AppDb, actor: SessionUser, userId: number, password: string): Promise<UserResult> {
  const check = await manageable(db, actor, userId);
  if (!check.ok) return check;
  const pe = passwordError(password);
  if (pe) return fail({ password: pe });
  await db.update(users).set({ passwordHash: await hashPassword(password), mustChangePassword: true }).where(eq(users.id, userId));
  await deleteUserSessions(db, userId);
  return OK;
}

export async function setActive(db: AppDb, actor: SessionUser, userId: number, active: boolean): Promise<UserResult> {
  const check = await manageable(db, actor, userId);
  if (!check.ok) return check;
  await db.update(users).set({ active }).where(eq(users.id, userId));
  if (!active) await deleteUserSessions(db, userId);
  return OK;
}

export async function changeOwnPassword(
  db: AppDb,
  userId: number,
  password: string,
  confirm: string,
  keepToken: string,
): Promise<UserResult> {
  const pe = passwordError(password);
  if (pe) return fail({ password: pe });
  if (password !== confirm) return fail({ confirm: 'Пароли не совпадают.' });
  await db.update(users).set({ passwordHash: await hashPassword(password), mustChangePassword: false }).where(eq(users.id, userId));
  await deleteUserSessions(db, userId, keepToken);
  return OK;
}

import { createHash } from 'node:crypto';
import { and, count, eq, gt, lt } from 'drizzle-orm';
import { loginAttempts, users } from '@/db/schema';
import type { AppDb } from '@/db/types';
import { normalizeLogin } from '@/lib/admin/users';
import { hashPassword, verifyPassword } from './password';
import { createSession, type SessionUser } from './session';

const WINDOW_MS = 15 * 60_000;
const MAX_FAILS = 5;
const KEEP_MS = 24 * 60 * 60_000;

export type LoginResult = { ok: true; token: string; user: SessionUser } | { ok: false; error: 'invalid' | 'locked' };

export const LOGIN_ERRORS: Record<'invalid' | 'locked', string> = {
  invalid: 'Неверный логин или пароль.',
  locked: 'Слишком много попыток. Попробуйте через 15 минут.',
};

// Если логина нет, всё равно считаем scrypt — время ответа не выдаёт, существует ли логин
let dummyHash: Promise<string> | null = null;

export async function login(
  db: AppDb,
  input: { login: string; password: string; ip: string; salt: string; now: Date },
): Promise<LoginResult> {
  const name = normalizeLogin(input.login);
  const now = input.now;
  const key = createHash('sha256').update(`${input.salt}:${name}:${input.ip}`).digest('hex');

  await db.delete(loginAttempts).where(lt(loginAttempts.createdAt, new Date(now.getTime() - KEEP_MS)));
  const [{ value }] = await db
    .select({ value: count() })
    .from(loginAttempts)
    .where(and(eq(loginAttempts.key, key), gt(loginAttempts.createdAt, new Date(now.getTime() - WINDOW_MS))));
  if (value >= MAX_FAILS) return { ok: false, error: 'locked' };

  const [u] = await db.select().from(users).where(eq(users.login, name));
  dummyHash ??= hashPassword('velaro-dummy-password');
  const passwordOk = await verifyPassword(input.password, u?.passwordHash ?? (await dummyHash));
  if (!u || !u.active || !passwordOk) {
    await db.insert(loginAttempts).values({ key, createdAt: now });
    return { ok: false, error: 'invalid' };
  }

  await db.delete(loginAttempts).where(eq(loginAttempts.key, key));
  await db.update(users).set({ lastLoginAt: now }).where(eq(users.id, u.id));
  const token = await createSession(db, u.id, now);
  return { ok: true, token, user: { id: u.id, login: u.login, name: u.name, role: u.role, mustChangePassword: u.mustChangePassword } };
}

import type { Role } from '@/db/schema';
import type { SessionUser } from './session';

export function authorize(user: SessionUser | null, role?: Role): 'ok' | 'login' | 'forbidden' {
  if (!user) return 'login';
  if (role === 'owner' && user.role !== 'owner') return 'forbidden';
  return 'ok';
}

/** Куда вернуть после входа: только страницы админки, иначе — список заявок */
export function safeNext(next: unknown): string {
  if (typeof next !== 'string') return '/admin';
  const inAdmin = next === '/admin' || next.startsWith('/admin/') || next.startsWith('/admin?');
  if (!inAdmin || next.includes('\\') || next.startsWith('/admin/login')) return '/admin';
  return next;
}

export function dateRange(from: string, to: string): string {
  const [, fm, fd] = from.split('-').map(Number);
  const [, tm, td] = to.split('-').map(Number);

  if (fm === tm) return `${fd}–${td} ${MONTHS[tm - 1]}`;
  return `${fd} ${MONTHS[fm - 1]} — ${td} ${MONTHS[tm - 1]}`;

import { and, count, desc, eq, ilike, isNull, like, or, type SQL } from 'drizzle-orm';
import { cars, requestEvents, requests, users } from '@/db/schema';
import type { AppDb } from '@/db/types';
import type { SessionUser } from '@/lib/auth/session';
import { adminUrl, formatRequestMessage } from '@/lib/telegram';
import { isStatus, STATUSES, type Status } from './statuses';

export const PAGE_SIZE = 50;

export type ListRow = {
  id: number; createdAt: Date; name: string; phone: string; carName: string;
  dateFrom: string; dateTo: string; total: number; status: string; assigneeName: string | null;
};
export type ListResult = {
  rows: ListRow[]; counts: Record<Status | 'all', number>; total: number; page: number; pages: number; status: Status | 'all';
};
export type RequestEvent = {
  id: number; userId: number; type: 'status' | 'assign' | 'note'; fromValue: string | null; toValue: string | null;
  text: string | null; createdAt: Date; authorName: string;
};
export type RequestDetail = typeof requests.$inferSelect & {
  carName: string; assigneeName: string | null; events: RequestEvent[]; people: Record<string, string>;
};
export type ActionResult = { ok: true } | { ok: false; error: string };

const OK: ActionResult = { ok: true };
const NOT_FOUND: ActionResult = { ok: false, error: 'Заявка не найдена.' };

const escapeLike = (s: string) => s.replace(/[\\%_]/g, (m) => `\\${m}`);

/** Поиск по имени (без учёта регистра) и по цифрам телефона; фильтр по машине */
function searchCondition(q?: string, car?: string): SQL | undefined {
  const conds: SQL[] = [];
  if (car) conds.push(eq(requests.carId, car));
  const text = q?.trim();
  if (text) {
    const byName = ilike(requests.name, `%${escapeLike(text)}%`);
    let digits = text.replace(/\D/g, '');
    if (digits.length === 11 && digits.startsWith('8')) digits = `7${digits.slice(1)}`;
    conds.push(digits.length >= 3 ? (or(byName, like(requests.phone, `%${digits}%`)) as SQL) : byName);
  }
  return conds.length ? and(...conds) : undefined;
}

export async function listRequests(
  db: AppDb,
  f: { status?: Status | 'all'; q?: string; car?: string; page?: number } = {},
): Promise<ListResult> {
  const base = searchCondition(f.q, f.car);
  const counts = Object.fromEntries([...STATUSES, 'all'].map((s) => [s, 0])) as Record<Status | 'all', number>;
  const grouped = await db.select({ status: requests.status, n: count() }).from(requests).where(base).groupBy(requests.status);
  for (const g of grouped) {
    if (isStatus(g.status)) counts[g.status] = g.n;
    counts.all += g.n;
  }

  const status = f.status ?? 'new';
  const total = counts[status];
  const pages = Math.max(1, Math.ceil(total / PAGE_SIZE));
  const page = Math.min(Math.max(1, Math.floor(f.page ?? 1) || 1), pages);
  const rows = await db
    .select({
      id: requests.id, createdAt: requests.createdAt, name: requests.name, phone: requests.phone, carName: cars.name,
      dateFrom: requests.dateFrom, dateTo: requests.dateTo, total: requests.total, status: requests.status, assigneeName: users.name,
    })
    .from(requests)
    .innerJoin(cars, eq(cars.id, requests.carId))
    .leftJoin(users, eq(users.id, requests.assignedTo))
    .where(status === 'all' ? base : and(base, eq(requests.status, status)))
    .orderBy(desc(requests.id))
    .limit(PAGE_SIZE)
    .offset((page - 1) * PAGE_SIZE);
  return { rows, counts, total, page, pages, status };
}

export async function getRequest(db: AppDb, id: number): Promise<RequestDetail | null> {
  if (!Number.isInteger(id) || id < 1) return null;
  const [row] = await db.select({ r: requests, carName: cars.name }).from(requests).innerJoin(cars, eq(cars.id, requests.carId)).where(eq(requests.id, id));
  if (!row) return null;
  const people = Object.fromEntries((await db.select({ id: users.id, name: users.name }).from(users)).map((u) => [String(u.id), u.name]));
  const events = await db
    .select({
      id: requestEvents.id, userId: requestEvents.userId, type: requestEvents.type, fromValue: requestEvents.fromValue,
      toValue: requestEvents.toValue, text: requestEvents.text, createdAt: requestEvents.createdAt, authorName: users.name,
    })
    .from(requestEvents)
    .innerJoin(users, eq(users.id, requestEvents.userId))
    .where(eq(requestEvents.requestId, id))
    .orderBy(desc(requestEvents.id));
  const assigneeName = row.r.assignedTo ? (people[String(row.r.assignedTo)] ?? null) : null;
  return { ...row.r, carName: row.carName, assigneeName, events, people };
}

/** Взять в работу: побеждает первый — условие assigned_to IS NULL в самом UPDATE */
export async function takeRequest(db: AppDb, id: number, actor: SessionUser, now: Date): Promise<ActionResult> {
  return db.transaction(async (tx): Promise<ActionResult> => {
    const [taken] = await tx
      .update(requests)
      .set({ assignedTo: actor.id })
      .where(and(eq(requests.id, id), isNull(requests.assignedTo)))
      .returning({ status: requests.status });
    if (!taken) {
      const [cur] = await tx.select({ name: users.name }).from(requests).leftJoin(users, eq(users.id, requests.assignedTo)).where(eq(requests.id, id));
      if (!cur) return NOT_FOUND;
      return { ok: false, error: `Заявку уже взял(а) ${cur.name}.` };
    }
    await tx.insert(requestEvents).values({ requestId: id, userId: actor.id, type: 'assign', fromValue: null, toValue: String(actor.id), createdAt: now });
    if (taken.status === 'new') {
      await tx.update(requests).set({ status: 'in_progress' }).where(eq(requests.id, id));
      await tx.insert(requestEvents).values({ requestId: id, userId: actor.id, type: 'status', fromValue: 'new', toValue: 'in_progress', createdAt: now });
    }
    return OK;
  });
}

export async function assignRequest(db: AppDb, id: number, actor: SessionUser, toUserId: number, now: Date): Promise<ActionResult> {
  const [r] = await db.select({ assignedTo: requests.assignedTo }).from(requests).where(eq(requests.id, id));
  if (!r) return NOT_FOUND;
  if (actor.role !== 'owner' && r.assignedTo !== actor.id) {
    return { ok: false, error: 'Передать заявку может только ответственный или владелец.' };
  }
  const [target] = await db.select({ active: users.active }).from(users).where(eq(users.id, toUserId));
  if (!target || !target.active) return { ok: false, error: 'Такого сотрудника нет или он отключён.' };
  if (r.assignedTo === toUserId) return OK;
  await db.update(requests).set({ assignedTo: toUserId }).where(eq(requests.id, id));
  await db.insert(requestEvents).values({
    requestId: id, userId: actor.id, type: 'assign',
    fromValue: r.assignedTo === null ? null : String(r.assignedTo), toValue: String(toUserId), createdAt: now,
  });
  return OK;
}

export async function setStatus(db: AppDb, id: number, actor: SessionUser, status: unknown, now: Date): Promise<ActionResult> {
  if (!isStatus(status)) return { ok: false, error: 'Неизвестный статус.' };
  const [r] = await db.select({ status: requests.status }).from(requests).where(eq(requests.id, id));
  if (!r) return NOT_FOUND;
  if (r.status === status) return OK;
  await db.update(requests).set({ status }).where(eq(requests.id, id));
  await db.insert(requestEvents).values({ requestId: id, userId: actor.id, type: 'status', fromValue: r.status, toValue: status, createdAt: now });
  return OK;
}

export async function addNote(db: AppDb, id: number, actor: SessionUser, text: unknown, now: Date): Promise<ActionResult> {
  const t = typeof text === 'string' ? text.trim() : '';
  if (!t) return { ok: false, error: 'Напишите текст заметки.' };
  if (t.length > 2000) return { ok: false, error: 'Заметка — до 2000 символов.' };
  const [r] = await db.select({ id: requests.id }).from(requests).where(eq(requests.id, id));
  if (!r) return NOT_FOUND;
  await db.insert(requestEvents).values({ requestId: id, userId: actor.id, type: 'note', text: t, createdAt: now });
  return OK;
}

function ask(question: string, hidden = false): Promise<string> {

  return (await cookies()).get(SESSION_COOKIE)?.value;

/** Для страниц и серверных функций: нет сессии → на вход, не хватает прав → 404 */
export async function requireUser(role?: Role, opts: { allowMustChange?: boolean } = {}): Promise<SessionUser> {
  const user = await currentUser();
  const verdict = authorize(user, role);
  if (verdict === 'login') {
    const path = (await headers()).get('x-admin-path') ?? '/admin';
    redirect(`/admin/login?next=${encodeURIComponent(path)}`);
  }
  if (verdict === 'forbidden') notFound();
  if (user!.mustChangePassword && !opts.allowMustChange) redirect('/admin/password');
  return user!;
}

import { SESSION_COOKIE, sessionCookieOptions } from '@/lib/auth/cookie';

// Для /admin: передаёт текущий путь страницам (для ?next=), продлевает cookie, запрещает индексацию
export function middleware(req: NextRequest) {
  const headers = new Headers(req.headers);
  headers.set('x-admin-path', req.nextUrl.pathname + req.nextUrl.search);
  const res = NextResponse.next({ request: { headers } });
  const token = req.cookies.get(SESSION_COOKIE)?.value;
  if (token) res.cookies.set(SESSION_COOKIE, token, sessionCookieOptions());
  res.headers.set('X-Robots-Tag', 'noindex, nofollow');
  return res;
}

    salt: process.env.IP_HASH_SALT ?? 'local-dev-salt',
    now: new Date(),
  });
  if (!res.ok) return { error: LOGIN_ERRORS[res.error], values: { login: loginValue } };

  redirect(res.user.mustChangePassword ? '/admin/password' : safeNext(fd.get('next')));
}

export async function logoutAction(): Promise<void> {
  const token = await sessionToken();
  if (token) await deleteSession(getDb(), token);
  (await cookies()).set(SESSION_COOKIE, '', { ...sessionCookieOptions(), maxAge: 0 });
  redirect('/admin/login');
}

export async function changePasswordAction(_prev: ActionState, fd: FormData): Promise<ActionState> {
  const user = await requireUser(undefined, { allowMustChange: true });
  const res = await changeOwnPassword(getDb(), user.id, str(fd, 'password'), str(fd, 'confirm'), (await sessionToken()) ?? '');
  if (!res.ok) return { errors: res.errors };
  redirect('/admin');
}

export async function createManagerAction(_prev: ActionState, fd: FormData): Promise<ActionState> {
  const owner = await requireUser('owner');
  const input = { login: str(fd, 'login'), name: str(fd, 'name'), password: str(fd, 'password') };
  const res = await createManager(getDb(), owner, input);
  revalidatePath('/admin/managers');
  if (!res.ok) return { errors: res.errors, values: { login: input.login, name: input.name } };
  return { ok: true };
}

export async function resetPasswordAction(userId: number, _prev: ActionState, fd: FormData): Promise<ActionState> {
  const owner = await requireUser('owner');
  const res = await resetPassword(getDb(), owner, userId, str(fd, 'password'));
  revalidatePath('/admin/managers');
  return res.ok ? { ok: true } : { error: Object.values(res.errors)[0] };
}

export async function setActiveAction(userId: number, active: boolean, _prev: ActionState, _fd: FormData): Promise<ActionState> {
  const owner = await requireUser('owner');
  const res = await setActive(getDb(), owner, userId, active);
  revalidatePath('/admin/managers');
  return res.ok ? { ok: true } : { error: Object.values(res.errors)[0] };
}
